{{/* Galería masonry con lightbox. Uso: {{< gallery images="a.jpg|Pie de foto, b.jpg" alt="texto por defecto" cols="3" >}} Un tercer campo «luto» (a.jpg|Pie|luto) agrega el moño de luto al pie de foto. */}} {{- $cols := .Get "cols" | default "3" -}} {{- $defaultAlt := .Get "alt" | default "" -}} {{- $group := printf "g-%d" .Ordinal -}}
{{- range split (.Get "images") "," -}} {{- $parts := split (trim . " ") "|" -}} {{- $name := trim (index $parts 0) " " -}} {{- $caption := "" -}}{{- if gt (len $parts) 1 -}}{{- $caption = trim (index $parts 1) " " -}}{{- end -}} {{- $luto := and (gt (len $parts) 2) (eq (trim (index $parts 2) " ") "luto") -}} {{- $label := cond (and $luto $caption) (printf "%s (en memoria)" $caption) $caption -}} {{- with $.Page.Resources.Get $name -}} {{- $big := partial "large.html" . -}} {{- else -}} {{- errorf "gallery: no se encontró la imagen %q en %s" $name $.Page.File.Path -}} {{- end -}} {{- end -}}
