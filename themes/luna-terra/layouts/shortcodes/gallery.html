" ") "|" -}} {{- $name := trim (index $parts 0) " " -}} {{- $caption := "" -}}{{- if gt (len $parts) 1 -}}{{- $caption = trim (index $parts 1) " " -}}{{- end -}} {{- with $.Page.Resources.Get $name -}} {{- $big := partial "large.html" . -}} {{- else -}} {{- errorf "gallery: no se encontró la imagen %q en %s" $name $.Page.File.Path -}} {{- end -}} {{- end -}}
